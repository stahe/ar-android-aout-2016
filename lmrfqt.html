    6.2. تثبيت برنامج تشغيل (driver) Arduino

    لكي يتمكن IDE من التواصل مع Arduino، يجب أن يتعرف عليه المضيف PC. وللقيام بذلك، يمكن اتباع الخطوات التالية:

    • توصيل Arduino بمنفذ USB في الكمبيوتر المضيف؛
    • سيحاول النظام عندئذ العثور على برنامج تشغيل الجهاز الجديد USB. ولن يتمكن من العثور عليه. قم عندئذ بالإشارة إلى أن برنامج التشغيل موجود على القرص في الموضع <arduino>/drivers حيث <arduino> هو مجلد تثبيت جهاز Arduino IDE.

    • توصيل Arduino بـ PC عبر كابل USB. في الوقت الحالي، لا تقم بتضمين بطاقة الشبكة؛
    • في [1]، اختر نوع بطاقة Arduino المتصلة بالمنفذ USB؛
    • في [2]، حدد المنفذ USB الذي تتصل به Arduino. لمعرفة ذلك، افصل Arduino وسجل المنافذ. أعد توصيلها وتحقق مرة أخرى من المنافذ: المنفذ الذي تمت إضافته هو منفذ Arduino؛

    6.5. اختبار تطبيق شبكة

    باستخدام IDE، قم بتحميل المثال [Exemples / Ethernet / WebServer]:

    
/*
  Web Server
 
 A simple web server that shows the value of the analog input pins.
 using an Arduino Wiznet Ethernet shield. 
 
 Circuit:
 * Ethernet shield attached to pins 10, 11, 12, 13
 * Analog inputs attached to pins A0 through A5 (optional)
 
 created 18 Dec 2009
 by David A. Mellis
 modified 9 Apr 2012
 by Tom Igoe
 
 */

#تضمين <SPI.h>
#تضمين <Ethernet.h>

// أدخل عنوان MAC وعنوان IP لوحدة التحكم الخاصة بك أدناه.
// سيعتمد عنوان IP على شبكتك المحلية:
byte mac[] = { 
  0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };
IPAddress ip(192,168,1, 177);

// قم بتهيئة مكتبة خادم إيثرنت
// باستخدام عنوان IP والمنفذ الذي تريد استخدامه 
// (المنفذ 80 هو الإعداد الافتراضي لـ HTTP):
EthernetServer server(80);

void setup() {
// افتح الاتصالات التسلسلية وانتظر حتى يفتح المنفذ:
  Serial.begin(9600);
   while (!Serial) {
    ; // انتظر حتى يتصل المنفذ التسلسلي. مطلوب لـ Leonardo فقط
  }


  // بدء اتصال إيثرنت والخادم:
  Ethernet.begin(mac, ip);
  server.begin();
  Serial.print("server is at ");
  Serial.println(Ethernet.localIP());
}


void loop() {
  // الاستماع إلى العملاء الواردين
  EthernetClient client = server.available();
  if (client) {
    Serial.println("new client");
    // ينتهي طلب http بسطر فارغ
    boolean currentLineIsBlank = true;
    while (client.connected()) {
      if (client.available()) {
        char c = client.read();
        Serial.write(c);
        // إذا وصلت إلى نهاية السطر (استلمت حرف
        // ) وكان السطر فارغًا، فإن طلب http قد انتهى،
        // لذا يمكنك إرسال رد
        if (c == '\n' && currentLineIsBlank) {
          // أرسل رأس استجابة HTTP قياسي
          client.println("HTTP/1.1 200 OK");
          client.println("Content-Type: text/html");
          client.println("Connection: close");
          client.println();
          client.println("<!DOCTYPE HTML>");
          client.println("<html>");
                    // أضف علامة meta refresh، بحيث يقوم المتصفح بالسحب مرة أخرى كل 5 ثوانٍ:
          client.println("<meta http-equiv=\"refresh\" content=\"5\">");
          // إخراج قيمة كل دبوس إدخال تناظري
          for (int analogChannel = 0; analogChannel < 6; analogChannel++) {
            int sensorReading = analogRead(analogChannel);
            client.print("analog input ");
            client.print(analogChannel);
            client.print(" is ");
            client.print(sensorReading);
            client.println("<br />");       
          }
          client.println("</html>");
          break;
        }
        if (c == '\n') {
          // أنت تبدأ سطرًا جديدًا
          currentLineIsBlank = true;
        } 
        else if (c != '\r') {
          // لقد حصلت على حرف في السطر الحالي
          currentLineIsBlank = false;
        }
      }
    }
    // إعطاء متصفح الويب الوقت الكافي لاستلام البيانات
    delay(1);
    // إغلاق الاتصال:
    client.stop();
    Serial.println("client disconnected");
  }
}

    يُنشئ هذا التطبيق خادم ويب على المنفذ 80 (السطر 30) على العنوان IP في السطر 25. العنوان MAC في السطر 23 هو العنوان MAC المحدد على بطاقة الشبكة الخاصة بـ Arduino.

    تقوم الدالة [setup] بتهيئة خادم الويب:

    • السطر 34: تهيئ المنفذ التسلسلي الذي ستقوم التطبيق بتسجيل السجلات عليه. سنقوم بمتابعة هذه السجلات؛
    • السطر 41: يتم تهيئة العقدة TCP-IP (IP، المنفذ)؛
    • السطر 42: يتم تشغيل الخادم الموجود في السطر 30 على هذه العقدة الشبكية؛
    • السطران 43-44: يتم تسجيل عنوان IP لخادم الويب؛

    تقوم الدالة [loop] بتنفيذ خادم الويب:

    • السطر 50: إذا اتصل عميل بخادم الويب، فإن [server].available تعرض هذا العميل، وإلا فإنها تعرض null؛
    • السطر 51: إذا لم يكن العميل هو null؛
    • السطر 55: طالما أن العميل متصل؛
    • السطر 56: تكون قيمة [client].available صحيحة إذا أرسل العميل أحرفًا. يتم تخزين هذه الأحرف في مخزن مؤقت. تكون قيمة [client].available صحيحة طالما أن هذا المخزن المؤقت غير فارغ؛
    • السطر 57: يتم قراءة حرف أرسله العميل؛
    • السطر 58: يتم عرض هذا الحرف في وحدة التحكم الخاصة بالسجلات؛
    • السطر 62: في بروتوكول HTTP، يتبادل العميل والخادم أسطر النص.
      • يرسل العميل طلبًا HTTP إلى خادم الويب عن طريق إرسال سلسلة من أسطر النص تنتهي بسطر فارغ،
      • ثم يرد الخادم على العميل بإرسال استجابة وإغلاق الاتصال؛

    السطر 62، لا يقوم الخادم بأي شيء مع الرؤوس HTTP التي يتلقاها من العميل. إنه ينتظر ببساطة السطر الفارغ: سطر يحتوي على الحرف \n وحده؛

    • الأسطر 64-67: يرسل الخادم إلى العميل أسطر النص القياسية لبروتوكول HTTP. تنتهي هذه الأسطر بسطر فارغ (السطر 67)؛
    • ابتداءً من السطر 68، يرسل الخادم مستندًا إلى عميله. يتم إنشاء هذا المستند ديناميكيًا بواسطة الخادم ويكون بتنسيق HTML (السطور 68-69)؛
    • السطر 71: سطر HTML خاص يطلب من متصفح العميل تحديث الصفحة كل 5 ثوانٍ. وبالتالي، سيطلب المتصفح نفس الصفحة كل 5 ثوانٍ؛
    • السطور 73-80: يرسل الخادم إلى متصفح العميل قيم المدخلات التناظرية الستة الخاصة بـ Arduino؛
    • السطر 81: يتم إغلاق المستند HTML؛
    • السطر 82: الخروج من حلقة while في السطر 55؛
    • السطر 97: يتم إغلاق الاتصال مع العميل؛
    • السطر 98: يتم تسجيل الحدث في وحدة التحكم في السجلات؛
    • السطر 100: نعود إلى بداية الدالة loop: سيستأنف الخادم الاستماع إلى العملاء. ذكرنا أن متصفح العميل سيطلب نفس الصفحة كل 5 ثوانٍ. وسيكون الخادم جاهزًا للرد عليه مرة أخرى.

    6.7. جهاز Android اللوحي

    تم اختبار الأمثلة باستخدام جهاز Samsung Galaxy Tab 2 اللوحي.

    لاختبار الأمثلة باستخدام جهاز لوحي، يجب عليك أولاً تثبيت برنامج تشغيل الجهاز على جهاز التطوير الخاص بك. يمكن العثور على برنامج تشغيل جهاز Samsung Galaxy Tab 2 في:

     

    لاختبار الأمثلة، ستحتاج إلى توصيل الجهاز اللوحي بشبكة (ربما شبكة واي فاي) ومعرفة عنوانه IP على هذه الشبكة. وإليك كيفية القيام بذلك (Samsung Galaxy Tab 2):

    • قم بتشغيل جهازك اللوحي؛
    • ابحث في التطبيقات المتاحة على الجهاز اللوحي (في أعلى اليمين) عن التطبيق الذي يسمى [paramètres] والذي يحمل رمز الترس؛
    • في القسم الأيسر، قم بتفعيل الواي فاي؛
    • في القسم الأيمن، حدد شبكة واي فاي؛
    • بمجرد الاتصال بالشبكة، اضغط مرة واحدة على الشبكة المحددة. سيتم عرض عنوان IP الخاص بالجهاز اللوحي. قم بتدوينه. ستحتاج إليه؛

    لا تزال في تطبيق [Paramètres]،

    • حدد على اليسار الخيار [Options de développement] (في أسفل الخيارات)؛
    • تأكد من تحديد الخيار [Débogage USB] على اليمين.

    للعودة إلى القائمة، اضغط في شريط الحالة في الأسفل على الرمز الأوسط، وهو رمز المنزل. وفي شريط الحالة في الأسفل أيضًا،

    • الرمز الموجود في أقصى اليسار هو رمز الرجوع للخلف: يعيدك إلى العرض السابق؛
    • أما الرمز الموجود في أقصى اليمين فهو رمز إدارة المهام. يمكنك عرض وإدارة جميع المهام التي تم تنفيذها في وقت معين بواسطة جهازك اللوحي؛

    قم بتوصيل جهازك اللوحي بـ PC باستخدام كابل USB المرفق معه.

    6.14. إدارة الإطار jSON في Java

    بشكل شفاف للمطور، يستخدم إطار العمل [Spring MVC] مكتبة jSON [Jackson]. لتوضيح ماهية jSON (JavaScript Object Notation)، نقدم هنا برنامجًا يقوم بتسلسل الكائنات في jSON ويقوم بالعكس عن طريق فك تسلسل سلاسل jSON التي تم إنتاجها لإعادة إنشاء الكائنات الأولية.

    تسمح مكتبة "Jackson" بإنشاء:

    • سلسلة jSON لكائن: new ObjectMapper().writeValueAsString(object)؛
    • كائن من سلسلة jSON: new ObjectMapper().readValue(jsonString, Object.class).

    الفئة [Main] هي كما يلي:

    
package istia.st.json;

import com.fasterxml.jackson.databind.ObjectMapper;

import java.io.IOException;
import java.util.HashMap;
import java.util.Map;

public class Main {
  // أداة التسلسل/إلغاء التسلسل
  static ObjectMapper mapper = new ObjectMapper();

  public static void main(String[] args) throws IOException {
    // إنشاء شخص
    Personne paul = new Personne("Denis", "Paul", 40);
    // عرض Json
    String json = mapper.writeValueAsString(paul);
    System.out.println("Json=" + json);
    // إنشاء مثيل Personne من Json
    Personne p = mapper.readValue(json, Personne.class);
    // عرض الشخص
    System.out.println("Personne=" + p);
    // جدول
    Personne virginie = new Personne("Radot", "Virginie", 20);
    Personne[] personnes = new Personne[]{paul, virginie};
    // عرض Json
    json = mapper.writeValueAsString(personnes);
    System.out.println("Json personnes=" + json);
    // قاموس
    Map<String, Personne> hpersonnes = new HashMap<String, Personne>();
    hpersonnes.put("1", paul);
    hpersonnes.put("2", virginie);
    // عرض Json
    json = mapper.writeValueAsString(hpersonnes);
    System.out.println("Json hpersonnes=" + json);
  }
}

    6.15. تثبيت [WampServer]

    [WampServer] عبارة عن مجموعة برامج للتطوير في PHP / MySQL / Apache على جهاز يعمل بنظام Windows. سنستخدمه فقط لـ SGBD MySQL.

    • على موقع [WampServer] [1]، اختر الإصدار المناسب [2]،
    • الملف القابل للتنفيذ الذي تم تنزيله هو برنامج تثبيت. يتم طلب معلومات متنوعة أثناء التثبيت. هذه المعلومات لا تتعلق بـ MySQL. لذا يمكن تجاهلها. تظهر نافذة [3] في نهاية التثبيت. قم بتشغيل [WampServer]،
     
    • في [4]، يتم تثبيت رمز [WampServer] في شريط المهام أسفل يمين الشاشة [4]،
    • عند النقر عليها، تظهر القائمة [5]. وهي تسمح بإدارة خادم Apache و SGBD MySQL. لإدارة هذا الخادم، نستخدم الخيار [PhpPmyAdmin]،
    • ثم تظهر النافذة أدناه،

    لن نقدم سوى القليل من التفاصيل حول استخدام [PhpMyAdmin]. نوضح في الوثيقة كيفية استخدامه.
